cmake_minimum_required(VERSION 3.20)
project(ULID LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.5" CACHE STRING "" FORCE)
endif()

add_library(ULID MODULE 4DPlugin-ULID.cpp "4D Plugin API/4DPluginAPI.c")
target_include_directories(ULID PRIVATE "4D Plugin API" suyash)
set_source_files_properties("4D Plugin API/4DPluginAPI.c" PROPERTIES LANGUAGE CXX)
set(bundle "${CMAKE_BINARY_DIR}/package/ULID.bundle/Contents")
file(MAKE_DIRECTORY "${bundle}/Resources")
configure_file(Info.plist "${bundle}/Info.plist" COPYONLY)
configure_file(manifest.json "${bundle}/Resources/manifest.json" COPYONLY)
configure_file(constants.xlf "${bundle}/Resources/constants.xlf" COPYONLY)
configure_file(LICENSE "${bundle}/Resources/LICENSE" COPYONLY)
configure_file("4D Plugin API/LICENSE" "${bundle}/Resources/LICENSE-4D" COPYONLY)
configure_file(suyash/LICENSE "${bundle}/Resources/LICENSE-suyash" COPYONLY)
if(APPLE)
  target_link_libraries(ULID PRIVATE "-framework CoreFoundation" "-framework CoreGraphics")
  set_target_properties(ULID PROPERTIES PREFIX "" SUFFIX ""
    LIBRARY_OUTPUT_DIRECTORY "${bundle}/MacOS")
elseif(WIN32)
  target_compile_definitions(ULID PRIVATE WIN32 WIN64 NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_sources(ULID PRIVATE "4D Plugin API/4DPluginAPI.def")
  target_link_libraries(ULID PRIVATE bcrypt)
  set_target_properties(ULID PROPERTIES PREFIX "" SUFFIX ".4DX"
    LIBRARY_OUTPUT_DIRECTORY "${bundle}/Windows64"
    RUNTIME_OUTPUT_DIRECTORY "${bundle}/Windows64")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(ULID PROPERTIES
      LIBRARY_OUTPUT_DIRECTORY_${config} "${bundle}/Windows64"
      RUNTIME_OUTPUT_DIRECTORY_${config} "${bundle}/Windows64")
  endforeach()
else()
  message(FATAL_ERROR "The 4D plug-in supports macOS and Windows only")
endif()

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(ulid_core_tests tests/core_tests.cpp)
  target_include_directories(ulid_core_tests PRIVATE "${CMAKE_SOURCE_DIR}")
  target_link_libraries(ulid_core_tests PRIVATE Threads::Threads)
  add_test(NAME ulid_core COMMAND ulid_core_tests)
  add_executable(ulid_boundary_tests tests/boundary_tests.cpp 4DPlugin-ULID.cpp)
  target_include_directories(ulid_boundary_tests PRIVATE "${CMAKE_SOURCE_DIR}" "4D Plugin API" suyash)
  if(APPLE)
    target_link_libraries(ulid_boundary_tests PRIVATE "-framework CoreFoundation")
  else()
    target_compile_definitions(ulid_boundary_tests PRIVATE WIN32 WIN64 NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(ulid_boundary_tests PRIVATE bcrypt)
  endif()
  add_test(NAME ulid_boundary COMMAND ulid_boundary_tests)
endif()
